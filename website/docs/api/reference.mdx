## Types

See [TypeScript](../guides/typescript#exported-types) for the full list. The most used:

```ts
interface DateRange { start: Date | null; end: Date | null }

interface MarkedDate { dots?: string[]; disabled?: boolean; label?: string; textColor?: string }

interface SelectOption<T> {
  label: string; value: T; description?: string; disabled?: boolean;
  icon?: React.ReactNode; group?: string; keywords?: string[];
}

interface WheelItem<T> { label: string; value: T; disabled?: boolean; key?: string }

type SkinTone = 0 | 1 | 2 | 3 | 4 | 5;

/** Every sheet's `presentation`. */
type SheetPresentation = 'auto' | 'bottomSheet' | 'modal';

/** SelectPicker's `presentation`; its `auto` is a dropdown on the web. */
type SelectPresentation = SheetPresentation | 'dropdown';

/** The field slots every `*PickerField` and `PickerField` takes in `styles`. */
type PickerFieldSlot = 'label' | 'trigger' | 'triggerText' | 'placeholder' | 'chevron' | 'clear';

type SlotStyles<K extends string> = Partial<Record<K, StyleProp<TextStyle>>>;

interface ThemableProps { theme?: DeepPartial<PickerTheme> }
```

## Default strings

| Where | Prop | Default |
| --- | --- | --- |
| Sheets | `confirmLabel` / `cancelLabel` | `Done` / `Cancel` |
| SelectPicker, fields | `placeholder` | `Select…` |
| SelectList | `searchPlaceholder` / `emptyText` | `Search` / `No results` |
| EmojiPicker | `searchPlaceholder` / `emptyText` | `Search emoji` / `No emoji found` |
| TimePicker | `dayPeriodLabels` | from locale (`AM` / `PM`) |
